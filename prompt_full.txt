/** Resample a rep's angle sequence to N evenly spaced points (linear interpolation). */
export function resampleCurve(frames, n = 16) {
  if (!frames || frames.length < 2) return null;
  const a = frames.map(f => f.angle);
  const out = [];
  for (let i = 0; i < n; i++) {
    const pos = (i / (n - 1)) * (a.length - 1);
    const lo = Math.floor(pos), hi = Math.min(a.length - 1, lo + 1);
    out.push(Math.round((a[lo] + (a[hi] - a[lo]) * (pos - lo)) * 10) / 10);
  }
  return out;
}

## PHASE 1 — Feature 32: Fatigue detection (frontend + stored summary)

### 1.1 New file `frontend/src/utils/fatigueDetection.js`
Pure, dependency-free, unit-testable. Export `computeFatigue(reps)`.
Input: array of rep objects `{ repNumber, durationMs, peakAngle, smoothnessScore, qualityScore }`.
Algorithm:
1. If `reps.length < 4` → `{ level:'none', index:0, signals:[], enoughData:false }`.
2. Baseline = first `max(2, floor(n/3))` reps (median of each metric); Recent = last `max(2, floor(n/3))` reps (median).
3. Signals (each only counts if the change exceeds noise):
   - `tempo`: `(recent.durationMs - base.durationMs)/base.durationMs` ≥ 0.20 → slowdown %.
   - `rom`: `(base.peakAngle - recent.peakAngle)/base.peakAngle` ≥ 0.08 → ROM drop %.
   - `smoothness`: `base.smoothnessScore - recent.smoothnessScore` ≥ 10 points.
   - `quality`: `base.qualityScore - recent.qualityScore` ≥ 10 points.
4. `index = clamp(0,100, 100*(0.30*clamp01(tempo/0.5) + 0.30*clamp01(rom/0.25) + 0.20*clamp01(smoothDrop/30) + 0.20*clamp01(qualityDrop/30)))`.
5. `level`: `<20 none`, `<45 mild`, `<70 moderate`, else `high`.
6. Return `{ level, index:Math.round(index), signals:[{code,label,value}], enoughData:true }` where labels are patient-friendly (e.g. "Reps are getting slower").
